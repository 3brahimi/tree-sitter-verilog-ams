module m(a);
  electrical a;
  real t, vt, x;
  analog begin
    t = $abstime;
    vt = $vt($temperature);
    x = $simparam("gmin", 1e-12);
    if ($param_given(t)) x = 1;
    if ($port_connected(a)) x = 2;
    $bound_step(1n);
    $discontinuity(0);
    x = $limit(V(a), "pnjlim", 0.6, 0.026);
    x = $mfactor;
    $strobe("t=%g", t);
  end
endmodule
